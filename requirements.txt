# cosmos-edge-serve — runtime dependencies, all pinned exactly.
#
# The CUDA 12.8 wheel index is required to get GPU-enabled torch on Windows.
# On Linux x86_64 the default PyPI torch wheels are already cu128, so this line
# is a no-op there. Harmless either way.
--extra-index-url https://download.pytorch.org/whl/cu128

# ---------------------------------------------------------------------------
# Torch stack. These four versions are locked together and must move together:
# torchvision 0.24.0 declares `torch==2.9.0` as a hard requirement, and
# torchcodec 0.9.x is built against torch 2.9. Bumping one alone will not resolve.
# torch 2.9.0 is the version NVIDIA pins in cosmos-reason2/scripts/inference_sample.py.
# ---------------------------------------------------------------------------
torch==2.9.0
torchvision==0.24.0
# torchcodec decodes video. No aarch64 wheels, so Jetson/Graviton fall back to `av`.
torchcodec==0.9.1; platform_machine != "aarch64"

# ---------------------------------------------------------------------------
# Model stack. transformers 4.57.3 is NVIDIA's pin and is the floor for
# Qwen3VLForConditionalGeneration (needs >= 4.57).
#
# huggingface-hub MUST stay below 1.0: transformers 4.57.3 declares
# `huggingface-hub<1.0,>=0.34.0`. Hub is on 1.x now, so an unpinned install
# silently resolves transformers backwards to a version without Qwen3-VL.
# 0.36.2 is the last 0.x release.
#
# tokenizers is capped at <=0.23.0 by transformers 4.57.3; 0.23.0 was never
# published, so 0.22.2 is the real ceiling.
# ---------------------------------------------------------------------------
transformers==4.57.3
huggingface-hub==0.36.2
tokenizers==0.22.2
safetensors==0.6.2
accelerate==1.12.0

# 4-bit NF4 quantization for the 6 GB dev laptop. CUDA only — has no effect on CPU.
bitsandbytes==0.48.2
# LoRA/QLoRA adapter loading from the featherweight-ai repo. Off by default.
peft==0.18.0

# Media decoding
pillow==12.0.0
av==16.1.0

# ---------------------------------------------------------------------------
# Service

fastapi==0.141.1
uvicorn[standard]==0.52.1
pydantic-settings==2.15.0
python-multipart==0.0.32
structlog==26.1.0
